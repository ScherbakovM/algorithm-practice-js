9. Перемещение нулей

    console.log(arr[i]);
  }
}

Фрагмент B:

function exampleB(arr) {
  for (let i = 0; i < arr.length; i++) {

    }
  }
}

function exampleC(n) {
  while (n > 1) {
    n = Math.floor(n / 2);
  }
}
